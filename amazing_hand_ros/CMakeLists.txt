cmake_minimum_required(VERSION 3.10)
project(amazing_hand_ros)
add_compile_options(-Wall -Wextra -Wpedantic)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(catkin REQUIRED COMPONENTS roscpp hardware_interface controller_manager transmission_interface urdf)
catkin_package(INCLUDE_DIRS include LIBRARIES amazing_hand_bus CATKIN_DEPENDS roscpp hardware_interface controller_manager transmission_interface urdf)
include_directories(include ${catkin_INCLUDE_DIRS})
add_library(amazing_hand_bus src/scs0009.cpp)
target_link_libraries(amazing_hand_bus ${catkin_LIBRARIES})
add_executable(amazing_hand_node src/amazing_hand_node.cpp)
target_link_libraries(amazing_hand_node amazing_hand_bus ${catkin_LIBRARIES})
install(TARGETS amazing_hand_bus amazing_hand_node ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY launch config urdf DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md LICENSE NOTICE DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  add_rostest(test/integration.test)
  catkin_add_gtest(test_driver test/test_driver.cpp)
  if(TARGET test_driver)
    target_compile_definitions(test_driver PRIVATE PACKAGE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(test_driver amazing_hand_bus ${catkin_LIBRARIES} util pthread)
  endif()
endif()

cmake_minimum_required(VERSION 3.10)
project(amazing_hand_driver)
install(FILES LICENSE NOTICE DESTINATION share/${PROJECT_NAME})

if("$ENV{ROS_VERSION}" STREQUAL "2")
  find_package(ament_cmake REQUIRED)
  find_package(feetech_driver REQUIRED)
  find_package(transmission_interface REQUIRED)
  find_package(hardware_interface REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(pluginlib REQUIRED)

  add_library(amazing_hand_hardware_interface SHARED src/amazing_hand_hardware_interface.cpp)
  target_compile_features(amazing_hand_hardware_interface PRIVATE cxx_std_20)
  target_link_libraries(amazing_hand_hardware_interface PRIVATE feetech_driver::communication_protocol
    hardware_interface::hardware_interface transmission_interface::transmission_interface
    pluginlib::pluginlib rclcpp::rclcpp)
  pluginlib_export_plugin_description_file(hardware_interface amazing_hand_hardware_interface.xml)
  install(TARGETS amazing_hand_hardware_interface LIBRARY DESTINATION lib)

  install(DIRECTORY launch/ DESTINATION share/${PROJECT_NAME}/launch FILES_MATCHING PATTERN "*.launch.xml")
  install(FILES config/ros2_controllers.yaml config/hand_ros2.rviz DESTINATION share/${PROJECT_NAME}/config)
  install(FILES urdf/amazing_hand.ros2_control.xacro DESTINATION share/${PROJECT_NAME}/urdf)
  install(PROGRAMS scripts/hand_calibration.py scripts/hand_description scripts/hand_rviz_config DESTINATION lib/${PROJECT_NAME})
  install(PROGRAMS scripts/hand_motion_demo_ros2 DESTINATION lib/${PROJECT_NAME})
  install(FILES scripts/hand_motion_poses.py DESTINATION lib/${PROJECT_NAME})
  if(BUILD_TESTING)
    find_package(ament_cmake_pytest REQUIRED)
    find_package(ament_cmake_ros REQUIRED)
    ament_add_ros_isolated_pytest_test(ros2_hardware test/test_ros2_hardware.py TIMEOUT 100)
  endif() # if(BUILD_TESTING)
  ament_package()
else() # if("$ENV{ROS_VERSION}" STREQUAL "2")
  add_compile_options(-Wall -Wextra -Wpedantic)
  set(CMAKE_CXX_STANDARD 14)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  find_package(catkin REQUIRED COMPONENTS gazebo_dev gazebo_ros_control roscpp hardware_interface controller_manager transmission_interface urdf)
  catkin_package(INCLUDE_DIRS include LIBRARIES amazing_hand_bus CATKIN_DEPENDS roscpp hardware_interface controller_manager transmission_interface urdf)
  include_directories(include ${catkin_INCLUDE_DIRS})
  add_library(amazing_hand_hw_sim SHARED src/amazing_hand_hw_sim.cpp)
  target_link_libraries(amazing_hand_hw_sim ${catkin_LIBRARIES})
  install(FILES amazing_hand_hw_sim.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
  add_library(amazing_hand_bus src/scs0009.cpp)
  target_link_libraries(amazing_hand_bus ${catkin_LIBRARIES})
  add_executable(amazing_hand_node src/amazing_hand_node.cpp)
  target_link_libraries(amazing_hand_node amazing_hand_bus ${catkin_LIBRARIES})
  install(TARGETS amazing_hand_hw_sim amazing_hand_bus amazing_hand_node ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
  install(DIRECTORY launch/ DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/launch FILES_MATCHING PATTERN "*.launch")
  install(FILES config/ros_controllers.yaml config/hand_ros1.rviz DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/config)
  install(FILES urdf/amazing_hand.urdf.xacro DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/urdf)
  install(PROGRAMS scripts/hand_calibration.py scripts/hand_rviz_config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/scripts)
  install(PROGRAMS scripts/hand_motion_demo_ros1 DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  install(FILES scripts/hand_motion_poses.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  if(CATKIN_ENABLE_TESTING)
    find_package(rostest REQUIRED)
    add_rostest(test/integration.test)
    catkin_add_gtest(test_driver test/test_driver.cpp)
    if(TARGET test_driver)
      target_compile_definitions(test_driver PRIVATE PACKAGE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
      target_link_libraries(test_driver amazing_hand_bus ${catkin_LIBRARIES} util pthread)
    endif() # if(TARGET test_driver)
  endif() # if(CATKIN_ENABLE_TESTING)
endif() # if("$ENV{ROS_VERSION}" STREQUAL "2")
